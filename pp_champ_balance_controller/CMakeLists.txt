cmake_minimum_required(VERSION 3.0.2)
project(pp_champ_balance_controller)
add_compile_options(-std=c++11 -Wall -Wextra)
find_package(catkin REQUIRED COMPONENTS roscpp geometry_msgs sensor_msgs tf2)
catkin_package(CATKIN_DEPENDS roscpp geometry_msgs sensor_msgs tf2)
include_directories(${catkin_INCLUDE_DIRS})
add_executable(pp_champ_balance_controller_node src/pp_champ_balance_controller_node.cpp)
target_link_libraries(pp_champ_balance_controller_node ${catkin_LIBRARIES})
add_dependencies(pp_champ_balance_controller_node ${catkin_EXPORTED_TARGETS})
install(TARGETS pp_champ_balance_controller_node RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(DIRECTORY launch/ config/ DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
